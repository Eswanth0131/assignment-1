class

        current_q 
=
 
self

self

self

            action

self
.actions)


        
else
:

self
.q_table[state]


            action 
=
 
self
.arg_max(state_action)

@staticmethod


    
def

        max_index_list 
=
 []


        max_value 
=
 state_action[
0
]


        
for
 index, value 
in
 
enumerate
(state_action):


            
if
 value 
>
 max_value:


                max_index_list.clear()


                max_value 
=
 value


                max_index_list.append(index)


            
elif
 value 
==
 max_value:


                max_index_list.append(index)


        
return
 random.choice(max_index_list)
